- [ ] **Step 1: Celebration.luau** :

```luau
local ReplicatedStorage = game:GetService("ReplicatedStorage")

local Gameplay = require(ReplicatedStorage.Shared.Gameplay)
local Fx = require(script.Parent.Fx)
local Audio = require(script.Parent.Audio)
local Cinematic = require(script.Parent.Cinematic)

local UiKit = require(script.Parent.UiKit)

local Celebration = {}

function Celebration.Tier(rank: number): string
	if rank >= Gameplay.RarityRank("UltraRare") then
		return "ultra"
	elseif rank >= Gameplay.RarityRank("Legendary") then
		return "legendary"
	elseif rank >= Gameplay.RarityRank("Epic") then
		return "epic"
	end
	return "none"
end

function Celebration.Hatch(rank: number)
	local tier = Celebration.Tier(rank)
	if tier == "none" then
		return
	end
	local pos = workspace.CurrentCamera.CFrame.Position - workspace.CurrentCamera.CFrame.LookVector * 12
	Fx.Confetti(pos, UiKit.TINTS.ACCENT, if tier == "epic" then 24 else 40)
	Audio.Play("RareFanfare")
	if tier ~= "epic" then
		Cinematic.Play({ Rarity = nil }) -- Cinematic.ShouldPlay gère le rang via payload.Rarity ; passer le rang réel
	end
end

return Celebration
```

(Ajuster : `Cinematic.Play` lit `payload.Rarity` — lui donner le rang directement ou exposer `Cinematic.PlayRank(rank)`.)
